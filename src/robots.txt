# AI crawler policy (#6820). Training and retrieval crawlers are controlled
# independently by every major AI company, so a single blanket rule can only
# get this half right. Recipe content is exactly what people ask assistants
# about, and retrieval bots are what let this site show up as a cited source
# in AI answers — blocking them removes the site from AI search entirely.

# Retrieval bots: fetch pages live to answer a specific user query and cite
# the source. Explicitly allowed by name rather than allowed only by
# omission, so a future blanket Disallow can't silently drop the site out of
# AI search results without someone noticing this group.
User-agent: OAI-SearchBot
Allow: /
Disallow: /api/
Disallow: /admin/
Disallow: /src/

Allow: /
Disallow: /api/
Disallow: /admin/
Disallow: /src/
